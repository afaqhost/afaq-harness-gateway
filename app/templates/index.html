curl http://localhost:3500/v1/models

إرسال رسالة

POST /v1/chat/completions

استبدل API Key والموديل بقيم موجودة في حسابك.

curl
curl http://localhost:3500/v1/chat/completions \
  -H "Authorization: Bearer afaq_YOUR_KEY" \
  -H "Content-Type: application/json" \
  -d '{"model":"opencode//opencode/big-pickle","messages":[{"role":"user","content":"Hello"}]}'
import os
import requests

response = requests.post(
    os.getenv("AFAQ_BASE_URL", "http://localhost:3500") + "/v1/chat/completions",
    headers={"Authorization": f"Bearer {os.environ['AFAQ_API_KEY']}"},
    json={"model": "opencode//opencode/big-pickle", "messages": [{"role": "user", "content": "Hello"}]},
)
response.raise_for_status()
print(response.json()["choices"][0]["message"]["content"])
const response = await fetch("http://localhost:3500/v1/chat/completions", { method: "POST", headers: { "Authorization": `Bearer ${process.env.AFAQ_API_KEY}`, "Content-Type": "application/json" }, body: JSON.stringify({ model: "opencode//opencode/big-pickle", messages: [{ role: "user", content: "Hello" }] }) });
const result = await response.json();
console.log(result.choices[0].message.content);
<?php
$ch = curl_init("http://localhost:3500/v1/chat/completions");
curl_setopt_array($ch, [CURLOPT_POST => true, CURLOPT_HTTPHEADER => ["Authorization: Bearer " . getenv("AFAQ_API_KEY"), "Content-Type: application/json"], CURLOPT_POSTFIELDS => json_encode(["model" => "opencode//opencode/big-pickle", "messages" => [["role" => "user", "content" => "Hello"]]]), CURLOPT_RETURNTRANSFER => true]);
$result = json_decode(curl_exec($ch), true);
echo $result["choices"][0]["message"]["content"];
?>

البث المباشر

stream: true

فعّل stream للحصول على أجزاء SSE تدريجيًا، وتنتهي الاستجابة الناجحة بـ data: [DONE].

أخطاء شائعة

401 يعني أن المصادقة فشلت (مفتاح غير صالح أو جلسة منتهية)، و400 يعني أن اسم الهارنس غير معروف، و502 يعني أن أداة CLI فشلت.
